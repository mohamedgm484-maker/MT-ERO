C

أم اللغات — أساس الأنظمة

يعني إيه C؟

C هي لغة برمجة إجرائية، بتعتبر أم كل اللغات الحديثة. بتستخدم في تطوير أنظمة التشغيل، الأدوات، والأنظمة المدمجة.

Linux و Windows و macOS كلهم مكتوبين بـ C. وكتير من اللغات التانية (C++, Java, Python) مبنية على C.

استخدامات C

  • أنظمة التشغيل
  • الأدوات والنظم
  • الأنظمة المدمجة
  • قواعد البيانات
  • الألعاب
  • الأدوات الأمنية

التثبيت

pkg install clang -y clang --version

الأساسيات

1. البرنامج الأول

#include <stdio.h> int main() { printf("Hello World\n"); return 0; }

2. المتغيرات

int age = 25; float salary = 5000.50; double pi = 3.14159; char letter = 'A'; char name[] = "Ahmed"; printf("%d\n", age); printf("%s\n", name); printf("%.2f\n", salary);

3. المدخلات

int age; printf("اكتب عمرك: "); scanf("%d", &age); printf("عمرك %d\n", age);

4. الشروط

int age = 20; if (age >= 18) { printf("بالغ\n"); } else if (age >= 13) { printf("مراهق\n"); } else { printf("طفل\n"); }

5. الحلقات

// for for (int i = 0; i < 5; i++) { printf("%d\n", i); } // while int count = 0; while (count < 5) { printf("%d\n", count); count++; }

6. الدوال

int add(int a, int b) { return a + b; } void greet(char name[]) { printf("أهلاً %s\n", name); } int main() { printf("%d\n", add(5, 3)); greet("Ahmed"); return 0; }

7. المصفوفات

int numbers[5] = {1, 2, 3, 4, 5}; for (int i = 0; i < 5; i++) { printf("%d\n", numbers[i]); } // String char name[] = "Ahmed"; printf("%s\n", name);

8. المؤشرات (Pointers)

int x = 10; int *ptr = &x; printf("القيمة: %d\n", x); printf("العنوان: %p\n", ptr); printf("القيمة من المؤشر: %d\n", *ptr);
